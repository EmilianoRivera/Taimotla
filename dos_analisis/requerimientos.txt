% --- BÚSQUEDA Y NAVEGACIÓN ---
	\FRitem{RF-28}{Búsqueda de expedientes por nombre}{El sistema debe permitir buscar expedientes de NNA en tiempo real escribiendo el nombre o apellido del menor en una barra de búsqueda.}{A}{RU-30}
	\FRitem{RF-29}{Búsqueda de expedientes por CURP}{El sistema debe permitir buscar expedientes ingresando la CURP completa o parcial del NNA.}{A}{RU-30}
	\FRitem{RF-30}{Búsqueda de expedientes por folio}{El sistema debe permitir localizar un expediente ingresando su número de folio o número de reporte consecutivo.}{A}{RU-30}
	\FRitem{RF-31}{Filtrado de expedientes por estado}{El sistema debe permitir filtrar la lista de expedientes según su estado: Detección, En Diagnóstico, En Seguimiento, Cierre Solicitado o Cerrado.}{A}{RU-30}
	\FRitem{RF-32}{Filtrado de expedientes por rango de fechas}{El sistema debe permitir filtrar expedientes por rango de fecha de apertura o de última actualización.}{M}{RU-30}
	\FRitem{RF-33}{Paginación de resultados de expedientes}{El sistema debe paginar los resultados de expedientes mostrando un máximo de 20 registros por página.}{M}{RU-01}

% --- REGISTRO DE NNA ---
	\FRitem{RF-34}{Captura de datos de domicilio con autocompletado SEPOMEX}{El sistema debe autocompletar colonia, municipio y estado al ingresar un código postal válido, consultando el catálogo SEPOMEX.}{A}{RU-34}
	\FRitem{RF-35}{Registro de idiomas del NNA}{El sistema debe permitir registrar uno o más idiomas que habla el NNA seleccionándolos del catálogo INALI y especificando el nivel de dominio (Nativo, Fluido, Básico, Solo comprende).}{A}{RU-31}
	\FRitem{RF-36}{Registro de idiomas del tutor}{El sistema debe permitir registrar los idiomas del tutor con el mismo mecanismo que el NNA, vinculando el registro al tutor correspondiente.}{A}{RU-31}
	\FRitem{RF-37}{Validación de formato CURP}{El sistema debe validar que la CURP capturada para el NNA, tutor o personal cumpla con la estructura oficial de 18 caracteres alfanuméricos establecida por el RENAPO.}{A}{RU-03}
	\FRitem{RF-38}{Prevención de expedientes duplicados}{El sistema debe verificar que no exista un expediente previo activo con la misma CURP del NNA antes de crear uno nuevo.}{A}{RU-03}
	\FRitem{RF-39}{Registro del apodo o nombre preferido del NNA}{El sistema debe permitir capturar el nombre o apodo con el que el NNA prefiere ser llamado, conforme a la Caja de Herramientas §4.3.6.}{A}{RU-03}
	\FRitem{RF-40}{Registro de nacionalidad y etnia del NNA}{El sistema debe permitir registrar la nacionalidad y el origen étnico o pueblo indígena del NNA.}{A}{RU-03}
	\FRitem{RF-41}{Indicador de documentos registrales del NNA}{El sistema debe registrar si el NNA cuenta con acta de nacimiento y si está registrado en el Registro Civil, mostrando estos indicadores en el expediente.}{A}{RU-07}
	\FRitem{RF-42}{Carga y visualización de acta de nacimiento}{El sistema debe permitir cargar el acta de nacimiento digitalizada en formato PDF, JPG o PNG con un límite de 5 MB.}{A}{RU-07}
	\FRitem{RF-43}{Carga y visualización de cartilla de vacunación}{El sistema debe permitir cargar la cartilla de vacunación del NNA en formato PDF, JPG o PNG y marcar si está completa o incompleta.}{A}{RU-10}

% --- DATOS MÉDICOS ---
	\FRitem{RF-44}{Registro del seguro médico del NNA}{El sistema debe permitir seleccionar el tipo de seguro médico del NNA (IMSS, ISSSTE, Bienestar, Sin seguro) y capturar el número de afiliación.}{A}{RU-10}
	\FRitem{RF-45}{Registro de padecimientos del NNA con catálogo CIE-10}{El sistema debe permitir registrar uno o más padecimientos del NNA seleccionándolos del catálogo CIE-10, indicando si son crónicos y si están controlados.}{A}{RU-33}
	\FRitem{RF-46}{Registro de padecimientos del tutor con catálogo CIE-10}{El sistema debe permitir registrar los padecimientos del tutor con el mismo mecanismo del NNA, vinculados al registro del tutor.}{A}{RU-33}
	\FRitem{RF-47}{Registro de discapacidades del NNA}{El sistema debe permitir registrar el tipo de discapacidad (Motriz, Sensorial, Intelectual, Psicosocial, Múltiple), grado de dependencia y si el NNA requiere o cuenta con aditamentos.}{A}{RU-32}
	\FRitem{RF-48}{Registro de datos de alimentación y descanso}{El sistema debe permitir capturar la descripción de la alimentación habitual del NNA y sus horarios de sueño, conforme al checklist de la Caja de Herramientas §4.3.3.}{M}{RU-10}
	\FRitem{RF-49}{Registro de fecha de última visita médica}{El sistema debe permitir registrar la fecha de la última consulta médica del NNA y si actualmente recibe atención.}{M}{RU-10}

% --- ENTORNO FAMILIAR ---
	\FRitem{RF-50}{Registro de la dinámica del hogar}{El sistema debe permitir al Trabajador Social capturar el sistema de residencia, sostén económico, relación con familia de origen, patrón de migración y participación comunitaria de la familia del NNA.}{A}{RU-50}
	\FRitem{RF-51}{Registro de redes de apoyo familiares y comunitarias}{El sistema debe permitir registrar en texto libre la descripción de las redes de apoyo disponibles para la familia del NNA.}{A}{RU-50}
	\FRitem{RF-52}{Registro del grado de negación del cuidador}{El sistema debe permitir seleccionar el grado de negación del cuidador principal (Alto, Medio, Bajo) con un campo de observaciones, conforme a la Caja de Herramientas §4.3.1.}{A}{RU-16}
	\FRitem{RF-53}{Registro del grado de afectación emocional del cuidador}{El sistema debe permitir seleccionar el grado de afectación emocional o física del cuidador (Alto, Medio, Bajo) con un campo de observaciones, conforme a la Caja de Herramientas §4.3.1.}{A}{RU-16}
	\FRitem{RF-54}{Registro de la fecha acordada para el plan de restitución}{El sistema debe permitir registrar la fecha pactada con la familia para presentarles el Plan de Restitución de Derechos.}{A}{RU-16}

% --- FAMILIOGRAMA ---

	\FRitem{RF-56}{Carga de imagen del familiograma}{El sistema debe permitir adjuntar una imagen digitalizada del familiograma elaborado durante la entrevista, en formato JPG, PNG o PDF.}{M}{RU-37}

% --- ENTREVISTA CON EL NNA ---
	\FRitem{RF-58}{Registro de observaciones de la entrevista directa con el NNA}{El sistema debe permitir al Trabajador Social o Psicólogo registrar la actitud general, tono de voz, expresión corporal y cambios emocionales observados durante la entrevista con el NNA, conforme a la Caja de Herramientas §4.3.6.}{A}{RU-49}
	\FRitem{RF-59}{Registro de personas a quienes el NNA teme y personas significativas}{El sistema debe permitir registrar en texto libre a quién dice temer el NNA y qué adulto le es significativo, conforme al formato §4.3.6 de la Caja de Herramientas.}{A}{RU-49}
	\FRitem{RF-60}{Registro del checklist de vida cotidiana del NNA}{El sistema debe permitir capturar las respuestas (Sí, No, No aplica) al checklist de vida cotidiana del NNA: escuela, alimentación, juego, deporte, amigos, sueño; conforme a §4.3.6 de la Caja de Herramientas.}{A}{RU-49}

	\FRitem{RF-106}{Registro automático de bitácora de cambios del expediente}{El sistema debe registrar automáticamente en una bitácora de auditoría cada modificación realizada en el expediente, indicando el usuario, fecha, hora y campo modificado.}{A}{RU-44}
	\FRitem{RF-107}{Visualización del historial del expediente}{El sistema debe mostrar al personal autorizado el historial cronológico completo de todas las acciones realizadas sobre el expediente.}{A}{RU-44}
	\FRitem{RF-108}{Registro del número de reporte consecutivo automático}{El sistema debe asignar automáticamente un número de reporte consecutivo al crear un registro de detección de caso, sin posibilidad de edición manual.}{A}{RU-11}

% --- DASHBOARD ---
	\FRitem{RF-109}{Dashboard del Director con indicadores globales}{El sistema debe mostrar al Director un tablero con el total de expedientes por estado (Detección, En Diagnóstico, En Seguimiento, Cierre Solicitado, Cerrados), número de equipos activos y alertas pendientes totales.}{A}{RU-45}
	\FRitem{RF-110}{Panel de casos asignados para el Coordinador}{El sistema debe mostrar al Coordinador un panel con sus expedientes activos agrupados por estado, con indicadores de alertas y valoraciones pendientes.}{A}{RU-01}
	\FRitem{RF-111}{Panel de casos del especialista}{El sistema debe mostrar a cada especialista (Abogado, Médico, Psicólogo, Trabajador Social) únicamente los expedientes del equipo al que pertenece, con indicadores de sus valoraciones pendientes.}{A}{RU-02}

% --- SEGURIDAD Y PERMISOS ---
	\FRitem{RF-112}{Control de acceso basado en rol (RBAC)}{El sistema debe restringir el acceso a módulos y acciones según el rol del usuario: Director, Coordinador, Abogado, Médico, Psicólogo, Trabajador Social.}{A}{RU-06}
	\FRitem{RF-113}{Restricción de edición por equipo asignado}{El sistema debe impedir que un especialista edite o visualice expedientes de equipos a los que no pertenece.}{A}{RU-07}
	\FRitem{RF-114}{Modo de solo lectura para expedientes cerrados}{El sistema debe cambiar automáticamente el expediente a modo de solo lectura una vez que su estado sea Cerrado, impidiendo cualquier modificación posterior.}{A}{RU-26}
	\FRitem{RF-115}{Confirmación antes de acciones destructivas}{El sistema debe solicitar confirmación explícita del usuario antes de ejecutar acciones irreversibles como eliminar equipos, deshabilitar cuentas o solicitar el cierre de un expediente.}{M}{RU-06}

% --- CIERRE Y REAPERTURA ---
	\FRitem{RF-116}{Rechazo de solicitud de cierre por el Director}{El sistema debe permitir al Director rechazar una solicitud de cierre de expediente registrando la observación de rechazo y devolviendo el expediente al estado En Seguimiento.}{A}{RU-52}
	\FRitem{RF-117}{Registro de justificación de cierre}{El sistema debe obligar al Coordinador a capturar una justificación textual al enviar la solicitud de cierre del expediente.}{A}{RU-26}
	\FRitem{RF-118}{Notificación al Coordinador sobre resolución de cierre}{El sistema debe notificar al Coordinador cuando el Director autorice o rechace la solicitud de cierre del expediente.}{M}{RU-26}

% --- INTEROPERABILIDAD Y EXPORTACIÓN ---
	\FRitem{RF-119}{Exportación del expediente completo en PDF}{El sistema debe permitir generar un PDF con el expediente completo del NNA incluyendo datos personales, hecho victimal, valoraciones, PRD y bitácoras de seguimiento.}{M}{RU-27}
	\FRitem{RF-120}{Visualización en línea de documentos adjuntos}{El sistema debe permitir visualizar directamente en el navegador los documentos adjuntos al expediente (acta de nacimiento, cartilla, evidencias) sin necesidad de descargarlos.}{M}{RU-07}
