\begin{UseCase}{CU-04}{Asignación de casos}{
	Permite al Coordinador de Caso asociar un expediente de NNA recién creado a un equipo multidisciplinario específico para iniciar las valoraciones y la ejecución del Plan de Restitución de Derechos.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Reyes Pasqualli Atzin Ariel}
	\UCitem{Supervisa}{\color{Gray}Ramírez Cuevas Emiliano}
	\UCitem{Actor}{\hyperlink{tCoordinador}{Coordinador}}
	\UCitem{Propósito}{Vincular formalmente la responsabilidad de un expediente de NNA a un equipo técnico interdisciplinario.}
	\UCitem{Entradas}{Selección del expediente del menor y selección del equipo multidisciplinario a asignar.}
	\UCitem{Origen}{Mouse/Teclado}
	\UCitem{Salidas}{Confirmación de la asignación del caso en la interfaz del sistema.}
	\UCitem{Destino}{Pantalla y base de datos relacional.}
	\UCitem{Precondiciones}{El Coordinador debe haber iniciado sesión y el expediente del menor debe estar registrado con anterioridad.}
	\UCitem{Postcondiciones}{El expediente cambia su estado a ``En Diagnóstico'' y sus datos quedan disponibles para consulta y edición de los especialistas asignados al equipo.}
	\UCitem{Errores}{
		\begin{description}
			\item[E1:] No se ha seleccionado un caso o no se ha seleccionado un equipo.
			\item[E2:] El usuario actual no cuenta con permisos de Coordinación.
		\end{description}
	}
	\UCitem{Tipo}{Caso de uso primario}
	\UCitem{Observaciones}{Regido por las reglas de negocio \hyperlink{BR-01}{BR-01} y \hyperlink{BR-02}{BR-02}.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Accede al panel de asignación de casos en el módulo de Coordinador.
	\UCpaso Muestra la lista de expedientes pendientes de asignación y la lista de equipos multidisciplinarios disponibles.
	\UCpaso[\UCactor] Selecciona el caso del menor de la lista.
	\UCpaso[\UCactor] Selecciona el equipo de trabajo al cual desea encomendar el caso.
	\UCpaso[\UCactor] Presiona el botón \IUbutton{Asignar caso}.
	\UCpaso Valida que se haya seleccionado tanto un menor como un equipo de especialistas. \Trayref{A}.
	\UCpaso Verifica que el Coordinador tenga permisos válidos en el sistema. \Trayref{B}.
	\UCpaso Asocia en la base de datos el ID del equipo con el expediente del NNA y actualiza el estado del expediente a ``En Diagnóstico''.
	\UCpaso Muestra el mensaje de éxito {\bf MSG-09} indicando que el caso fue asignado exitosamente al equipo.
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}

\begin{UCtrayectoriaA}{A}{Falta selección de caso o de equipo}
	\UCpaso Muestra el mensaje de error {\bf MSG-04} alertando que debe seleccionar un expediente y un equipo multidisciplinario obligatoriamente.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Regresa al paso 3 de la trayectoria principal.
\end{UCtrayectoriaA}

\begin{UCtrayectoriaA}{B}{Usuario no autorizado}
	\UCpaso Muestra el mensaje de error {\bf MSG-02} indicando que no cuenta con el rol de Coordinador de Caso.
	\UCpaso Termina el caso de uso.
\end{UCtrayectoriaA}
