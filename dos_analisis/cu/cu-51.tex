\begin{UseCase}{CU-51}{Registrar vida cotidiana y recreación del NNA}{
	Permite capturar el horario de juego, actividades recreativas, deportes practicados y si el NNA se junta con amigos.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Ramírez Cuevas Emiliano}
	\UCitem{Supervisa}{\color{Gray}Reyes Pasqualli Atzin Ariel}
	\UCitem{Actor}{\hyperlink{tTSocial}{Trabajador Social}}
	\UCitem{Propósito}{Documentar la vida cotidiana del NNA para evaluar la vulneración del derecho al descanso y esparcimiento.}
	\UCitem{Entradas}{Horario de juego, actividades a las que juega, práctica de deporte (booleano y cuál), frecuencia de convivencia con amigos, indicador si ha dejado de ver a alguien que quiere.}
	\UCitem{Origen}{Teclado y mouse.}
	\UCitem{Salidas}{Datos de vida cotidiana almacenados en el expediente.}
	\UCitem{Destino}{Pestaña Datos Familiares, subsección Vida cotidiana.}
	\UCitem{Precondiciones}{Usuario pertenece al equipo asignado.}
	\UCitem{Postcondiciones}{Datos registrados.}
	\UCitem{Errores}{Ninguno específico.}
	\UCitem{Tipo}{Caso de uso primario}
	\UCitem{Observaciones}{Conforme al checklist §4.3.3 y §4.3.6 de la Caja de Herramientas.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Accede a la subsección \IUbutton{Vida Cotidiana} dentro de Datos Familiares.
	\UCpaso Despliega el formulario de actividades cotidianas.
	\UCpaso[\UCactor] Captura el horario de juego, actividades, práctica de deporte y convivencia con amigos.
	\UCpaso[\UCactor] Indica si el NNA ha dejado de ver a alguien que quiere.
	\UCpaso[\UCactor] Presiona \IUbutton{Guardar vida cotidiana}.
	\UCpaso Almacena los datos y despliega \hyperlink{mMSG09}{\bf MSG-09}.
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}
