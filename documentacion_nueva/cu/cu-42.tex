\begin{UseCase}{CU-42}{Visualizar documentos adjuntos del expediente}{
	Permite al personal autorizado visualizar en línea, directamente en el navegador, los documentos adjuntos al expediente del NNA sin necesidad de descargarlos.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Ramírez Cuevas Emiliano}
	\UCitem{Supervisa}{\color{Gray}Rivera Segura José Emiliano}
	\UCitem{Actor}{\hyperlink{tDirector}{Director}, \hyperlink{tCoordinador}{Coordinador}, \hyperlink{tAbogado}{Abogado}, \hyperlink{tMedico}{Médico}, \hyperlink{tPsicologo}{Psicólogo}, \hyperlink{tTSocial}{Trabajador Social}}
	\UCitem{Propósito}{Agilizar la consulta documental sin requerir la descarga de archivos al equipo local del usuario.}
	\UCitem{Entradas}{Selección del documento a visualizar.}
	\UCitem{Origen}{Mouse.}
	\UCitem{Salidas}{Visor en línea del documento seleccionado.}
	\UCitem{Destino}{Panel de visor de documentos en la pantalla del expediente.}
	\UCitem{Precondiciones}{El usuario debe pertenecer al equipo del expediente o ser Director.}
	\UCitem{Postcondiciones}{El documento se muestra en línea sin ser descargado.}
	\UCitem{Errores}{\begin{description}\item[E1:] El archivo no puede cargarse en el visor (formato incompatible para visualización en línea).\end{description}}
	\UCitem{Tipo}{Caso de uso secundario}
	\UCitem{Observaciones}{Ninguna.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Accede a la pestaña \IUbutton{Documentos} del expediente y presiona el ícono de vista del documento.
	\UCpaso Carga el archivo desde el servidor de archivos. \Trayref{A}.
	\UCpaso Muestra el documento en el visor en línea integrado en la pantalla.
	\UCpaso[\UCactor] Revisa el documento y presiona \IUbutton{Cerrar visor} para volver a la lista.
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}

\begin{UCtrayectoriaA}{A}{Archivo no compatible con el visor}
	\UCpaso Muestra el mensaje \textbf{MSG-03} y ofrece la opción de descarga directa.
	\UCpaso[\UCactor] Oprime \IUbutton{Descargar}.
	\UCpaso Inicia la descarga del archivo.
\end{UCtrayectoriaA}
