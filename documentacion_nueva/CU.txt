%-------------------------------------- CU-19
\begin{UseCase}{CU-19}{Recuperar contraseña}{
	Permite a cualquier usuario registrado en el sistema solicitar el restablecimiento de su contraseña mediante el envío de un enlace de recuperación a su correo institucional registrado.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Rivera Segura José Emiliano}
	\UCitem{Supervisa}{\color{Gray}Ramírez Cuevas Emiliano}
	\UCitem{Actor}{\hyperlink{tDirector}{Director}, \hyperlink{tCoordinador}{Coordinador}, \hyperlink{tAbogado}{Abogado}, \hyperlink{tMedico}{Médico}, \hyperlink{tPsicologo}{Psicólogo}, \hyperlink{tTSocial}{Trabajador Social}}
	\UCitem{Propósito}{Restablecer el acceso al sistema de un usuario que olvidó sus credenciales sin requerir intervención del administrador.}
	\UCitem{Entradas}{Correo electrónico institucional registrado.}
	\UCitem{Origen}{Teclado}
	\UCitem{Salidas}{Correo electrónico con enlace de restablecimiento enviado al usuario.}
	\UCitem{Destino}{Pantalla de confirmación y bandeja de correo del usuario.}
	\UCitem{Precondiciones}{El usuario debe contar con una cuenta activa y un correo institucional registrado en el sistema.}
	\UCitem{Postcondiciones}{El usuario recibe un enlace temporal de un solo uso para restablecer su contraseña.}
	\UCitem{Errores}{
		\begin{description}
			\item[E1:] El correo ingresado no corresponde a ninguna cuenta registrada.
			\item[E2:] El enlace de recuperación ha expirado o ya fue utilizado.
		\end{description}
	}
	\UCitem{Tipo}{Caso de uso secundario}
	\UCitem{Observaciones}{Regido por la regla de negocio \hyperlink{BR-02}{BR-02}.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Presiona el enlace \IUbutton{¿Olvidaste tu contraseña?} en la pantalla de inicio de sesión.
	\UCpaso Muestra el formulario de recuperación solicitando el correo electrónico institucional.
	\UCpaso[\UCactor] Ingresa su correo y presiona el botón \IUbutton{Enviar enlace de recuperación}.
	\UCpaso Verifica que el correo corresponda a una cuenta registrada y activa en el sistema. \Trayref{A}.
	\UCpaso Genera un token de restablecimiento de un solo uso con expiración de 60 minutos.
	\UCpaso Envía un correo electrónico al usuario con el enlace de restablecimiento y muestra el mensaje de confirmación \textbf{MSG-09}.
	\UCpaso[\UCactor] Accede al enlace desde su correo, ingresa y confirma la nueva contraseña, y presiona \IUbutton{Restablecer contraseña}.
	\UCpaso Valida que el token sea válido y no haya expirado. \Trayref{B}.
	\UCpaso Actualiza la contraseña del usuario con el hash de la nueva clave e invalida el token.
	\UCpaso Redirige al usuario a la pantalla de inicio de sesión con el mensaje \textbf{MSG-09}.
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}

\begin{UCtrayectoriaA}{A}{Correo no registrado en el sistema}
	\UCpaso Muestra el mensaje de error \textbf{MSG-02} indicando que el correo no corresponde a ninguna cuenta activa.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Regresa al paso 3 de la trayectoria principal.
\end{UCtrayectoriaA}

\begin{UCtrayectoriaA}{B}{Token expirado o ya utilizado}
	\UCpaso Muestra el mensaje de error \textbf{MSG-02} indicando que el enlace de recuperación ha expirado o ya fue utilizado.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Redirige al paso 1 de la trayectoria principal para solicitar un nuevo enlace.
\end{UCtrayectoriaA}

%-------------------------------------- CU-20
\begin{UseCase}{CU-20}{Cambiar contraseña}{
	Permite a cualquier usuario autenticado actualizar su contraseña de acceso al sistema desde la pantalla de su perfil de cuenta.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Ramírez Cuevas Emiliano}
	\UCitem{Supervisa}{\color{Gray}Rivera Segura José Emiliano}
	\UCitem{Actor}{\hyperlink{tDirector}{Director}, \hyperlink{tCoordinador}{Coordinador}, \hyperlink{tAbogado}{Abogado}, \hyperlink{tMedico}{Médico}, \hyperlink{tPsicologo}{Psicólogo}, \hyperlink{tTSocial}{Trabajador Social}}
	\UCitem{Propósito}{Permitir al usuario actualizar su contraseña de forma segura sin requerir intervención del administrador.}
	\UCitem{Entradas}{Contraseña actual, nueva contraseña y confirmación de nueva contraseña.}
	\UCitem{Origen}{Teclado}
	\UCitem{Salidas}{Mensaje de confirmación del cambio exitoso.}
	\UCitem{Destino}{Pantalla de perfil de cuenta.}
	\UCitem{Precondiciones}{El usuario debe haber iniciado sesión con credenciales válidas.}
	\UCitem{Postcondiciones}{La contraseña del usuario queda actualizada en la base de datos y la sesión activa se mantiene.}
	\UCitem{Errores}{
		\begin{description}
			\item[E1:] La contraseña actual ingresada es incorrecta.
			\item[E2:] La nueva contraseña y su confirmación no coinciden.
			\item[E3:] La nueva contraseña no cumple con los requisitos mínimos de seguridad.
		\end{description}
	}
	\UCitem{Tipo}{Caso de uso secundario}
	\UCitem{Observaciones}{Ninguna.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Accede a la opción \IUbutton{Cambiar Contraseña} desde la pantalla de su perfil de cuenta.
	\UCpaso Despliega el formulario con campos para la contraseña actual, nueva contraseña y confirmación.
	\UCpaso[\UCactor] Ingresa la contraseña actual, la nueva contraseña y la confirmación, y presiona \IUbutton{Guardar cambios}.
	\UCpaso Verifica que la contraseña actual ingresada coincida con la almacenada. \Trayref{A}.
	\UCpaso Verifica que la nueva contraseña y su confirmación sean idénticas. \Trayref{B}.
	\UCpaso Verifica que la nueva contraseña cumpla con al menos 8 caracteres, incluyendo mayúsculas, minúsculas y un número. \Trayref{C}.
	\UCpaso Actualiza la contraseña del usuario con el hash de la nueva clave.
	\UCpaso Despliega el mensaje de éxito \textbf{MSG-09} confirmando el cambio.
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}

\begin{UCtrayectoriaA}{A}{Contraseña actual incorrecta}
	\UCpaso Muestra el mensaje de error \textbf{MSG-04} indicando que la contraseña actual ingresada es incorrecta.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Regresa al paso 3 de la trayectoria principal.
\end{UCtrayectoriaA}

\begin{UCtrayectoriaA}{B}{Nueva contraseña y confirmación no coinciden}
	\UCpaso Muestra el mensaje de error \textbf{MSG-04} indicando que los campos de nueva contraseña no coinciden.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Regresa al paso 3 de la trayectoria principal.
\end{UCtrayectoriaA}

\begin{UCtrayectoriaA}{C}{Contraseña no cumple requisitos de seguridad}
	\UCpaso Muestra el mensaje de error \textbf{MSG-04} indicando los requisitos mínimos de la contraseña.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Regresa al paso 3 de la trayectoria principal.
\end{UCtrayectoriaA}

\begin{UseCase}{CU-21}{Buscar y filtrar expedientes}{
	Permite a cualquier usuario autenticado localizar expedientes de NNA mediante búsqueda por nombre, CURP o folio, y filtrarlos por estado del expediente o rango de fechas desde el panel principal.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Rivera Segura José Emiliano}
	\UCitem{Supervisa}{\color{Gray}Guerra Salinas Edgar Rafael}
	\UCitem{Actor}{\hyperlink{tDirector}{Director}, \hyperlink{tCoordinador}{Coordinador}, \hyperlink{tAbogado}{Abogado}, \hyperlink{tMedico}{Médico}, \hyperlink{tPsicologo}{Psicólogo}, \hyperlink{tTSocial}{Trabajador Social}}
	\UCitem{Propósito}{Agilizar la localización de expedientes específicos dentro del panel de casos, reduciendo el tiempo de consulta.}
	\UCitem{Entradas}{Texto de búsqueda (nombre, CURP o folio), filtro de estado y rango de fechas (opcionales).}
	\UCitem{Origen}{Teclado y mouse.}
	\UCitem{Salidas}{Lista filtrada de expedientes que coinciden con los criterios ingresados.}
	\UCitem{Destino}{Panel de expedientes del rol activo.}
	\UCitem{Precondiciones}{El usuario debe haber iniciado sesión. Los especialistas solo ven expedientes de su equipo asignado.}
	\UCitem{Postcondiciones}{El sistema despliega la lista de resultados sin modificar ningún dato del expediente.}
	\UCitem{Errores}{
		\begin{description}
			\item[E1:] No se encontraron expedientes que coincidan con los criterios de búsqueda.
		\end{description}
	}
	\UCitem{Tipo}{Caso de uso secundario}
	\UCitem{Observaciones}{Regido por la regla de negocio \hyperlink{BR-04}{BR-04}.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Accede al panel de expedientes desde el menú de navegación principal.
	\UCpaso Muestra la lista de expedientes disponibles según el rol y equipo del usuario, con la barra de búsqueda y los controles de filtrado.
	\UCpaso[\UCactor] Ingresa un texto en la barra de búsqueda (nombre, CURP o folio) y/o selecciona filtros de estado y rango de fechas.
	\UCpaso Actualiza en tiempo real la lista mostrando los expedientes que coinciden con los criterios ingresados. \Trayref{A}.
	\UCpaso[\UCactor] Selecciona el expediente deseado de los resultados y el sistema lo redirige al detalle del expediente ({\hyperlink{CU-03}{CU-03}}).
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}

\begin{UCtrayectoriaA}{A}{Sin resultados para los criterios de búsqueda}
	\UCpaso Muestra el mensaje informativo \textbf{MSG-01} indicando que no se encontraron expedientes que coincidan con los criterios ingresados.
	\UCpaso[\UCactor] Modifica los criterios de búsqueda o limpia los filtros.
	\UCpaso Regresa al paso 3 de la trayectoria principal.
\end{UCtrayectoriaA}

\begin{UseCase}{CU-22}{Registrar datos médicos del NNA}{
	Permite al Médico del equipo multidisciplinario capturar el historial clínico inicial del NNA en el expediente: seguro médico, padecimientos (catálogo CIE-10), cartilla de vacunación, discapacidades, aditamentos requeridos y datos de alimentación y descanso.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Ramírez Cuevas Emiliano}
	\UCitem{Supervisa}{\color{Gray}Rivera Segura José Emiliano}
	\UCitem{Actor}{\hyperlink{tMedico}{Médico}}
	\UCitem{Propósito}{Documentar de forma estructurada el estado de salud del NNA para identificar derechos de salud vulnerados e insumar el PRD.}
	\UCitem{Entradas}{Tipo de seguro médico, número de afiliación, fecha de última consulta, indicador de cartilla de vacunación (completa/incompleta/no tiene), padecimientos (CIE-10, crónico, controlado), tipo y grado de discapacidad, requerimiento de aditamentos, descripción de alimentación y horarios de sueño.}
	\UCitem{Origen}{Teclado y mouse.}
	\UCitem{Salidas}{Datos médicos del NNA registrados en el expediente y mensaje de confirmación.}
	\UCitem{Destino}{Pantalla del expediente único, pestaña Datos Médicos.}
	\UCitem{Precondiciones}{El Médico debe pertenecer al equipo asignado al expediente y el expediente debe estar en estado En Diagnóstico o En Seguimiento.}
	\UCitem{Postcondiciones}{Los datos de salud del NNA quedan almacenados y el indicador de valoración médica se actualiza a completada en el expediente.}
	\UCitem{Errores}{
		\begin{description}
			\item[E1:] El usuario no pertenece al equipo asignado al expediente.
			\item[E2:] Campos obligatorios de salud vacíos (tipo de seguro médico).
		\end{description}
	}
	\UCitem{Tipo}{Caso de uso primario}
	\UCitem{Observaciones}{Regido por las reglas de negocio \hyperlink{BR-03}{BR-03} y \hyperlink{BR-04}{BR-04}.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Selecciona la pestaña \IUbutton{Datos Médicos} dentro del expediente único del NNA.
	\UCpaso Verifica que el Médico pertenezca al equipo asignado al expediente. \Trayref{A}.
	\UCpaso Despliega el formulario de datos médicos con las secciones de seguro médico, padecimientos, cartilla, discapacidades y vida cotidiana.
	\UCpaso[\UCactor] Selecciona el tipo de seguro médico del NNA y captura el número de afiliación si aplica.
	\UCpaso[\UCactor] Registra la fecha de la última visita médica e indica si el NNA recibe atención actualmente.
	\UCpaso[\UCactor] Marca el estado de la cartilla de vacunación y registra las vacunas faltantes si la cartilla está incompleta.
	\UCpaso[\UCactor] Agrega los padecimientos del NNA seleccionándolos del catálogo CIE-10, marcando si son crónicos y si están controlados. Presiona \IUbutton{Agregar padecimiento} por cada uno.
	\UCpaso[\UCactor] Registra las discapacidades del NNA indicando tipo y grado de dependencia, y si requiere o cuenta con aditamentos.
	\UCpaso[\UCactor] Captura la descripción de la alimentación habitual y los horarios de sueño del NNA.
	\UCpaso[\UCactor] Presiona el botón \IUbutton{Guardar Datos Médicos}.
	\UCpaso Valida que los campos obligatorios del formulario estén completos. \Trayref{B}.
	\UCpaso Almacena los datos médicos en las tablas correspondientes y actualiza el indicador de valoración médica en el expediente.
	\UCpaso Despliega el mensaje de éxito \textbf{MSG-09} confirmando el guardado.
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}

\begin{UCtrayectoriaA}{A}{Médico no pertenece al equipo asignado}
	\UCpaso Muestra el mensaje de error \textbf{MSG-02} indicando falta de permisos de edición.
	\UCpaso Termina la ejecución del caso de uso.
\end{UCtrayectoriaA}

\begin{UCtrayectoriaA}{B}{Campos obligatorios vacíos}
	\UCpaso Muestra el mensaje de error \textbf{MSG-04} indicando los campos requeridos pendientes.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Regresa al paso 4 de la trayectoria principal manteniendo los datos capturados.
\end{UCtrayectoriaA}

}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Ramírez Cuevas Emiliano}
	\UCitem{Supervisa}{\color{Gray}Rivera Segura José Emiliano}
	\UCitem{Actor}{Sistema (proceso automático)}
	\UCitem{Propósito}{Proteger la información confidencial de los expedientes ante situaciones de abandono no intencional de la sesión.}
	\UCitem{Entradas}{Tiempo de última actividad del usuario en sesión.}
	\UCitem{Origen}{Proceso automático del servidor de sesión.}
	\UCitem{Salidas}{Sesión expirada y redirección a pantalla de inicio de sesión con mensaje de aviso.}
	\UCitem{Destino}{Pantalla de inicio de sesión.}
	\UCitem{Precondiciones}{El usuario debe tener una sesión activa con inactividad mayor a 30 minutos.}
	\UCitem{Postcondiciones}{La sesión queda destruida y el usuario debe autenticarse nuevamente.}
	\UCitem{Errores}{Ninguno.}
	\UCitem{Tipo}{Caso de uso secundario}
	\UCitem{Observaciones}{Se muestra un mensaje de advertencia 2 minutos antes del cierre automático.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso El sistema detecta que el usuario lleva 28 minutos sin registrar actividad.

	\UCpaso[\UCactor] Si el usuario retoma la actividad, el contador se reinicia. De lo contrario, el sistema continúa al paso 4.
	\UCpaso Transcurridos los 30 minutos totales, destruye la sesión del usuario.
	\UCpaso Redirige al usuario a la pantalla de inicio de sesión mostrando el mensaje de sesión expirada.
	\UCpaso[] -- -- Fin del caso de uso.
