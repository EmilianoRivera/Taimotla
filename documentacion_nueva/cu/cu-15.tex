\begin{UseCase}{CU-15}{Registrar seguimiento de medidas de protección}{
	Permite a los integrantes del equipo multidisciplinario actualizar el estatus de ejecución de cada medida de protección especial contenida en el PRD vigente del expediente, capturando bitácoras de avance, evidencias y fechas de cumplimiento para su monitoreo continuo.
}
	\UCitem{Versión}{\color{Gray}1.0}
	\UCitem{Autor}{\color{Gray}Reyes Pasqualli Atzin Ariel}
	\UCitem{Supervisa}{\color{Gray}Guerra Salinas Edgar Rafael}
	\UCitem{Actor}{\hyperlink{tCoordinador}{Coordinador}, \hyperlink{tAbogado}{Abogado}, \hyperlink{tMedico}{Médico}, \hyperlink{tPsicologo}{Psicólogo}, \hyperlink{tTSocial}{Trabajador Social}}
	\UCitem{Propósito}{Garantizar la continuidad y trazabilidad del cumplimiento de cada medida de protección especial registrada en el PRD, evitando lagunas temporales en el seguimiento de la restitución de derechos.}
	\UCitem{Entradas}{Identificador de la medida de protección, estatus de avance (Pendiente, En proceso, Cumplida, No cumplida), fecha de actualización, observaciones y evidencia digital opcional.}
	\UCitem{Origen}{Teclado, mouse y archivos digitales.}
	\UCitem{Salidas}{Bitácora de seguimiento actualizada y mensaje de confirmación.}
	\UCitem{Destino}{Pantalla y base de datos relacional.}
	\UCitem{Precondiciones}{El expediente del NNA debe encontrarse en estatus ``En Seguimiento'' con un PRD vigente y el usuario debe pertenecer al equipo asignado al expediente.}
	\UCitem{Postcondiciones}{La bitácora de la medida de protección queda actualizada y, si todas las medidas alcanzan el estatus ``Cumplida'', el expediente queda habilitado para su cierre.}
	\UCitem{Errores}{
		\begin{description}
			\item[E1:] Intentar registrar seguimiento en un expediente que no tiene un PRD vigente.
			\item[E2:] Campos obligatorios de la bitácora vacíos o con formato inválido.
			\item[E3:] Carga de evidencia digital que no cumple con la regla de negocio \hyperlink{BR-10}{BR-10}.
		\end{description}
	}
	\UCitem{Tipo}{Caso de uso primario}
	\UCitem{Observaciones}{Regido por las reglas de negocio \hyperlink{BR-05}{BR-05} y \hyperlink{BR-10}{BR-10}.}
\end{UseCase}

\begin{UCtrayectoria}
	\UCpaso[\UCactor] Accede al expediente del NNA desde su panel de casos asignados y selecciona la pestaña \IUbutton{Seguimiento de Medidas}.
	\UCpaso Verifica que el expediente cuente con un PRD vigente. \Trayref{A}.
	\UCpaso Despliega el listado de medidas de protección especial del PRD, agrupadas por derecho vulnerado, con su estatus actual y fecha límite.
	\UCpaso[\UCactor] Selecciona una medida de protección y presiona el botón \IUbutton{Actualizar Seguimiento}.
	\UCpaso Muestra el formulario de bitácora con campos para el nuevo estatus, fecha de actualización, observaciones y carga de evidencia digital.
	\UCpaso[\UCactor] Captura la información de avance y, opcionalmente, adjunta un archivo de evidencia.
	\UCpaso[\UCactor] Presiona el botón \IUbutton{Guardar Bitácora}.
	\UCpaso Valida que los campos obligatorios de la bitácora estén completos. \Trayref{B}.
	\UCpaso Valida el formato y tamaño del archivo de evidencia digital cargado. \Trayref{C}.
	\UCpaso Guarda el registro en la bitácora de la medida y actualiza su estatus.
	\UCpaso Verifica si todas las medidas del PRD alcanzaron el estatus ``Cumplida'' y, de ser así, habilita la opción de cierre de expediente para el Coordinador.
	\UCpaso Despliega el mensaje de éxito {\bf MSG-09} confirmando el registro de la bitácora.
	\UCpaso[] -- -- Fin del caso de uso.
\end{UCtrayectoria}

\begin{UCtrayectoriaA}{A}{Expediente sin PRD vigente}
	\UCpaso Muestra el mensaje de advertencia {\bf MSG-03} indicando que el expediente no cuenta con un Plan de Restitución vigente.
	\UCpaso Termina la ejecución del caso de uso.
\end{UCtrayectoriaA}

\begin{UCtrayectoriaA}{B}{Campos obligatorios de la bitácora vacíos}
	\UCpaso Muestra el mensaje de error {\bf MSG-04} indicando los campos requeridos.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Mantiene la información capturada y regresa al paso 6 de la trayectoria principal.
\end{UCtrayectoriaA}

\begin{UCtrayectoriaA}{C}{Fallo en archivo de evidencia cargado}
	\UCpaso Muestra el mensaje de error {\bf MSG-04} indicando que el archivo no cuenta con una extensión compatible o supera el peso permitido.
	\UCpaso[\UCactor] Oprime el botón \IUbutton{Aceptar}.
	\UCpaso Regresa al paso 6 de la trayectoria principal para corregir el archivo adjunto.
\end{UCtrayectoriaA}
